function calcResults(){

scores.sort((a,b)=>b.score-a.score);

const tops=scores.filter(s=>s.score===max);
const isHybrid=tops.length>1;
const dominant=tops[0];
const second=tops.length>1?tops[1]:scores[1];

const nomeAncora=isHybrid?`${tops[0].name} + ${tops[1].name}`:dominant.name;
document.getElementById('dominant-name').textContent=nomeAncora;

document.getElementById('courses-list').innerHTML=recs.courses.map(c=>`<li><strong>${c.name}</strong> — ${c.why}</li>`).join('');

}

function drawRadar(canvasId,scores,size){
const c=document.getElementById(canvasId);c.width=size;c.height=size;
const ctx=c.getContext('2d');const cx=size/2,cy=size/2,r=size*0.38;
const n=8;const sorted=[...anchors.map(a=>scores.find(s=>s.id===a.id))];
ctx.clearRect(0,0,size,size);
// grid
for(let ring=1;ring<=6;ring++){
ctx.beginPath();
for(let i=0;i<=n;i++){const angle=(Math.PI*2*i/n)-Math.PI/2;ctx.lineTo(cx+r*(ring/6)*Math.cos(angle),cy+r*(ring/6)*Math.sin(angle))}
ctx.closePath();ctx.strokeStyle='#e5e7eb';ctx.stroke();
}
// data
ctx.beginPath();
sorted.forEach((s,i)=>{const angle=(Math.PI*2*i/n)-Math.PI/2;const v=s.score/30;const x=cx+r*v*Math.cos(angle);const y=cy+r*v*Math.sin(angle);i===0?ctx.moveTo(x,y):ctx.lineTo(x,y)});
ctx.closePath();ctx.fillStyle='rgba(0,87,217,0.15)';ctx.fill();ctx.strokeStyle='#0057D9';ctx.lineWidth=2;ctx.stroke();
// dots & labels
sorted.forEach((s,i)=>{const angle=(Math.PI*2*i/n)-Math.PI/2;const v=s.score/30;const x=cx+r*v*Math.cos(angle);const y=cy+r*v*Math.sin(angle);
ctx.beginPath();ctx.arc(x,y,4,0,Math.PI*2);ctx.fillStyle='#0057D9';ctx.fill();
const lx=cx+(r+20)*Math.cos(angle);const ly=cy+(r+20)*Math.sin(angle);
ctx.font='9px Poppins';ctx.fillStyle='#374151';ctx.textAlign='center';ctx.fillText(s.id,lx,ly+3);
});
}

function drawBar(canvasId,scores,size){
const c=document.getElementById(canvasId);c.width=size;c.height=size;
const ctx=c.getContext('2d');const sorted=[...anchors.map(a=>scores.find(s=>s.id===a.id))];
const max=Math.max(...sorted.map(s=>s.score));
const barW=28,gap=10,startX=20,startY=size-30;
sorted.forEach((s,i)=>{
const h=(s.score/30)*(size-60);const x=startX+i*(barW+gap);
ctx.fillStyle=s.score===max?'#F4B400':'#0057D9';
ctx.beginPath();ctx.roundRect(x,startY-h,barW,h,4);ctx.fill();
ctx.font='bold 10px Poppins';ctx.fillStyle='#374151';ctx.textAlign='center';
ctx.fillText(s.score,x+barW/2,startY-h-6);
ctx.font='9px Poppins';ctx.fillText(s.id,x+barW/2,startY+14);
});
}
